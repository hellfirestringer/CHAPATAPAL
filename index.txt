<body>
    <div class="game-container" id="stage">
        <div class="bg-layer"></div>
        <div class="bg-overlay"></div>
        <div class="game-interface">
            <!-- Название жестко зафиксировано сверху по центру -->
            <div class="top-block">
                <h1 class="header-title" id="hotline-text">Продуктовый<br>магазин</h1>
            </div>
            <!-- Счётчики адекватно и стильно опущены в нижнюю треть экрана -->
            <div class="stats-dashboard">
                <div class="stat-card left"><span class="stat-lbl">УДАРЫ</span><span class="stat-num" id="score-val">0</span></div>
                <div class="stat-card right"><span class="stat-lbl">СКИДКА</span><span class="stat-num"><span id="discount-val">0</span>%</span></div>
            </div>
        </div>
    </div>
    <script>
        const textContainer = document.getElementById('hotline-text'), textStr = textContainer.innerHTML;
        textContainer.innerHTML = ''; let delay = 0;
        textStr.split('<br>').forEach((part, index) => {
            part.split('').forEach(char => {
                const span = document.createElement('span'); span.innerHTML = char === ' ' ? '&nbsp;' : char;
                span.style.animationDelay = `${delay}s`; delay += .08; textContainer.appendChild(span);
            });
            if (index === 0) textContainer.appendChild(document.createElement('br'));
        });

        const stage = document.getElementById('stage'), scoreVal = document.getElementById('score-val'), discountVal = document.getElementById('discount-val');
        let score = localStorage.getItem('gse_score') ? parseInt(localStorage.getItem('gse_score')) : 0;
        
        function updateMatchStats() { 
            scoreVal.textContent = score.toLocaleString(); 
            let d = Math.floor(score / 200); 
            discountVal.textContent = d > 10 ? 10 : d; 
        }
        updateMatchStats();

        function createDecorItem() {
            const item = document.createElement('div'); item.className = 'decor-item';
            const size = Math.floor(Math.random() * 20) + 26; item.style.width = item.style.height = size + 'px'; item.style.opacity = (Math.random() * .4 + .2).toFixed(2);
            const startX = Math.random() * (stage.clientWidth - 46); let currentY = -120, currentRotation = Math.random() * 360;
            const speed = Math.random() * 1.5 + 2, rotationSpeed = (Math.random() * 2 - 1) * 2;
            item.style.transform = `translate3d(${startX}px, ${currentY}px, 0) rotate(${currentRotation}deg)`; stage.appendChild(item);
            function fall() {
                if (!stage.contains(item)) return; currentY += speed; currentRotation += rotationSpeed;
                item.style.transform = `translate3d(${startX}px, ${currentY}px, 0) rotate(${currentRotation}deg)`;
                if (currentY > stage.clientHeight + 120) item.remove(); else requestAnimationFrame(fall);
            }
            requestAnimationFrame(fall);
        }

        function createTarget(type = 'target') {
            const el = document.createElement('div'), isBoss = (type === 'boss'), isMegaStone = (type === 'mega_stone'), isKroks = (type === 'kroks');
            if (isBoss) el.className = 'shoe-item boss-shoe'; else if (isMegaStone) el.className = 'mega-stonek'; else if (isKroks) el.className = 'kroks-danger'; else el.className = 'shoe-item target-shoe';
            
            const sizeOffset = isKroks ? 160 : (isMegaStone ? 180 : (isBoss ? 160 : 112));
            const w = stage.clientWidth || window.innerWidth;
            const h = stage.clientHeight || window.innerHeight;
            
            const randX = Math.random() * (w - sizeOffset);
            // Кроксы теперь спавнятся строго по центру, не залезая на нижнюю плашку счета
            const randY = isKroks ? (Math.random() * (h - sizeOffset - 360) + 160) : -190;
            const currentRotation = Math.random() * 360;
            
            el.style.left = randX + 'px';
            el.style.top = randY + 'px';
            
            if (!isKroks) {
                var speed = isMegaStone ? (h + 200) / 300 : (isBoss ? (Math.random() * 3 + 4.5) : (Math.random() * 2 + 3.5));
                var rotationSpeed = isMegaStone ? .5 : (isBoss ? (Math.random() * 2 - 1) * 3 : (Math.random() * 2 - 1) * 2);
                el.style.transform = `rotate(${currentRotation}deg)`;
            }
            
            stage.appendChild(el);
            if (isKroks) { setTimeout(() => { if (stage.contains(el)) el.remove(); }, 4000); }

            el.addEventListener('click', e => {
                e.stopPropagation();
                if (isKroks) { score = 0; localStorage.setItem('gse_score', 0); updateMatchStats(); el.remove(); return; }
                const points = isMegaStone ? 10 : (isBoss ? 5 : 1); score += points; localStorage.setItem('gse_score', score); updateMatchStats();
                const rect = stage.getBoundingClientRect(), clickX = e.clientX - rect.left, clickY = e.clientY - rect.top;
                const punch = document.createElement('div'); punch.className = isMegaStone ? 'punch-effect punch-ultra' : (isBoss ? 'punch-effect punch-mega' : 'punch-effect');
                punch.style.left = clickX + 'px'; punch.style.top = clickY + 'px'; stage.appendChild(punch); setTimeout(() => punch.remove(), 400);
                const plus = document.createElement('div'); plus.className = 'floating-plus'; plus.textContent = `+${points}`; if (isBoss || isMegaStone) plus.style.color = '#ff5500';
                plus.style.left = clickX + 'px'; plus.style.top = (clickY - 25) + 'px'; stage.appendChild(plus); setTimeout(() => plus.remove(), 400); el.remove();
            });

            var fallY = randY;
            function fall() {
                if (!stage.contains(el) || isKroks) return; fallY += speed; el.style.top = fallY + 'px';
                if (fallY > h + 190) el.remove(); else requestAnimationFrame(fall);
            }
            requestAnimationFrame(fall);
        }

        setInterval(createDecorItem, 450);
        function loopSpawn(type, minTime, maxTime) { createTarget(type); setTimeout(() => loopSpawn(type, minTime, maxTime), Math.random() * (maxTime - minTime) + minTime); }
        setTimeout(() => loopSpawn('target', 1200, 2500), 1000);
        setTimeout(() => loopSpawn('boss', 10000, 25000), 5000);
        setTimeout(() => loopSpawn('mega_stone', 35000, 65000), 15000);
        
        function loopSpawnKroks() { createTarget('kroks'); setTimeout(loopSpawnKroks, 30000); }
        setTimeout(loopSpawnKroks, 15000);
    </script>
